export default {
  async fetch(req) {
    const target = new URL(req.url).searchParams.get('url');
    if (!target) return new Response('?url= gerekli', { status: 400 });
    const res = await fetch(target, { headers: { 'User-Agent': 'Mozilla/5.0' } });
    const body = await res.text();
    return new Response(body, { status: res.status, headers: { 'Access-Control-Allow-Origin': '*', 'Content-Type': res.headers.get('Content-Type') || 'application/json' } });
  }
};
